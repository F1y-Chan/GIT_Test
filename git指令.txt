git init                                                      初始化本地库
git remote add origin  http/                 绑定github远程仓库
git add *                                                  提交代码到缓存区
git commit -m “备注”                         提交倒库
git push origin  仓库名字                       推送到github

git reset --mixed HEAD~1                     撤销上一次提交，文件修改会保留在工作区
